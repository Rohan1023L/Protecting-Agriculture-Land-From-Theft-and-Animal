from picamera2 import Picamera2
import cv2
from flask import Flask, Response, jsonify, request  # <-- request added
import shutil
from gtts import gTTS
import psutil
from flask_cors import CORS 
import os
import subprocess

app = Flask(__name__)
CORS(app)


def speak_message(message):
    try:
        tts = gTTS(text=message, lang='en', slow=False)
        tts_file = "/tmp/voice.mp3"
        tts.save(tts_file)
        os.system(f"mpg123 {tts_file}")  # Play audio
        print(f"[TTS] Spoken: {message}")
    except Exception as e:
        print(f"[TTS ERROR] {e}")

# ---------------- Camera Section ----------------
try:
    picam2 = Picamera2()
    picam2.start()
    camera_connected = True
except Exception as e:
    print("Camera not connected:", e)
    picam2 = None
    camera_connected = False

def generate_frames():
    while True:
        if not camera_connected:
            break
        frame = picam2.capture_array()
        ret, buffer = cv2.imencode('.jpg', frame)
        frame_bytes = buffer.tobytes()
        yield (b'--frame\r\n'
               b'Content-Type: image/jpeg\r\n\r\n' + frame_bytes + b'\r\n')

@app.route('/alert', methods=['POST', 'GET'])
def alert():
    alert_type = request.form.get("type") or request.args.get("type")
    message = request.form.get("message") or request.args.get("message")

    if message:
        tts_message = message  # custom message if provided
    else:
        if alert_type == "person":
            tts_message = "Person entered the farm"
        elif alert_type == "dog":
            tts_message = "Dog entered the farm"
        elif alert_type == "cow":
            tts_message = "Cow entered the farm"
        elif alert_type:
            tts_message = f"{alert_type} detected in your farm"
        else:
            return "No alert type given", 400

    print(f"[ALERT] Speaking: {tts_message}")
    speak_message(tts_message)  # <-- replaced espeak with gTTS
    return "Alert spoken", 200

    
@app.route('/speak', methods=['POST'])
def speak():
    data = request.get_json()
    message = data.get("message", "").strip()

    if not message:
        return jsonify({"error": "Message cannot be empty"}), 400

    speak_message(message)
    return jsonify({"status": "spoken", "message": message}), 200



@app.route('/PALFAT')
def video_feed():
    if camera_connected:
        return Response(generate_frames(),
                        mimetype='multipart/x-mixed-replace; boundary=frame')
    else:
        return "Camera not connected", 503

# ---------------- Temperature Section ----------------
@app.route("/temperature")
def temperature():
    temp = os.popen("vcgencmd measure_temp").readline().replace("temp=","").strip()
    return jsonify({"temperature": temp})

#------------------ SD Card Details ----------------------
@app.route("/storage")
def storage():
    total, used, free = shutil.disk_usage("/")
    return jsonify({
        "total_GB": round(total / (1024**3), 2),
        "used_GB": round(used / (1024**3), 2),
        "free_GB": round(free / (1024**3), 2)
    })
    
# ------------------ Storage Ram ---------------------------
@app.route("/ram")
def ram_usage():
    mem = psutil.virtual_memory()
    return jsonify({
        "total_GB": round(mem.total / (1024**3), 2),
        "used_GB": round(mem.used / (1024**3), 2),
        "free_GB": round(mem.available / (1024**3), 2),
        "percent_used": mem.percent
    })
